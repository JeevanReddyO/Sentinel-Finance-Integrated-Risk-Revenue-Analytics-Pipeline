# ====================================================================
# INSTALLATION NOTES
# ====================================================================
# 
# 1. Create virtual environment (recommended):
#    python -m venv venv
#    # On Windows: venv\Scripts\activate
#    # On macOS/Linux: source venv/bin/activate
#
# 2. Upgrade pip:
#    python -m pip install --upgrade pip
#
# 3. Install dependencies:
#    pip install -r requirements.txt
#
# 4. Copy and configure environment variables:
#    cp .env.example .env
#    # Edit .env with your database credentials
#
# 5. Initialize database:
#    psql -U postgres -d bank_db -f database/schema.sql
#    psql -U postgres -d bank_db -f database/seed_data.sql
#
# 6. Run system diagnostics:
#    python src/run_diagnostics.py
#
# 7. Execute ETL pipeline:
#    python src/extract.py
#
# 8. Launch dashboard:
#    streamlit run src/app.py
#
# ====================================================================
# TROUBLESHOOTING
# ====================================================================
#
# If psycopg2 installation fails on macOS:
#   brew install postgresql
#   pip install psycopg2-binary
#
# If you encounter SSL/database connection issues:
#   Update DATABASE_URL in .env with proper SSL mode:
#   postgresql://user:pass@host:5432/database?sslmode=require
#
# For Supabase PostgreSQL:
#   Get connection string from Supabase dashboard
#   DATABASE_URL="postgresql://user:pass@host:5432/postgres?sslmode=require"
